private static void SwitchLanguageFromGame(string gameLanguage)
{
    if (string.IsNullOrEmpty(gameLanguage))
    {
        LocalizedHelper.CurrentLanguage = "En";
        return;
    }

    // PT: Normaliza para lowercase e corta a 2 letras (ex: "Portuguese" -> "pt")
    // EN: Normalize to lowercase and truncate to 2 letters (e.g. "Portuguese" -> "pt")
    var normalized = gameLanguage.ToLowerInvariant().Trim();
    if (normalized.Length > 2)
        normalized = normalized.Substring(0, 2);

    switch (normalized)
    {
        case "ch":
            LocalizedHelper.CurrentLanguage = "Zh";
            break;
        default:
            // PT: TryGetValue nunca lança excecao; fallback seguro para "En"
            // EN: TryGetValue never throws; safe fallback to "En"
            if (!LocalizedHelper.LanguageNamesDictionary.TryGetValue(normalized, out var lang))
                lang = "En";
            LocalizedHelper.CurrentLanguage = lang;
            break;
    }
}
